X_set

sail-arm's definition a64.sail

val X_set : forall 'n 'width,
  'width in {32, 64}.
  (int('n), int('width), bits('width)) -> unit

function X_set (n, width, value_name) = {
    assert(n >= 0 & n <= 31);
    assert(width == 32 | width == 64);
    if n != 31 then {
        _R(n) = ZeroExtend(value_name, 64)
    };
    return()
}
