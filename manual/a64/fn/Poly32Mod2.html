Poly32Mod2

sail-arm's definition a64.sail

val Poly32Mod2 : forall 'N, 'N > 32. (bits('N), bits(32)) -> bits(32)

function Poly32Mod2 (data_in, poly) = {
    assert('N > 32);
    data : bits('N) = data_in;
    foreach (i from ('N - 1) to 32 by 1 in dec) {
        if [data[i]] == 0b1 then {
            data[i - 1 .. 0] = EOR(data[i - 1 .. 0], poly @ Zeros(i - 32))
        };
        ()
    };
    return(data[31 .. 0])
}
