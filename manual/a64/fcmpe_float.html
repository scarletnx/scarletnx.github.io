match ftype {
  0b00 => {
      datasize = 32
  },
  0b01 => {
      datasize = 64
  },
  0b10 => {
      throw(Error_Undefined())
  },
  0b11 => {
      if HaveFP16Ext() then {
          datasize = 16
      } else {
          throw(Error_Undefined())
      }
