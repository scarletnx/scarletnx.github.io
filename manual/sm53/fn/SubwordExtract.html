function SubwordExtract(word : bits(32), bit_offset : bits(32), width : bits(32), size : int) -> bits(32) =
    if MOD(size, 2) == 1 then Bits32(BitsS(word, bit_offset, width)) else BitsU(word, bit_offset, width)
