Operation Sail

OUT: a geometry program's vertex emitted (bit 39) and its primitive cut (bit 40) on the stream that the register at 20, the constant buffer word or the immediate names (its low two bits). Each stream the program uses is a selection of its own; stream 0's vertex is the vertex stages' end (transform feedback, the position's fix, the primitive's number), another's its transform feedback's capture. d becomes the next output's handle, 0

let stream0 = match Form() { 2 => ZeroExtend(Field(20, 2), 32), 1 => CbufWord(), _ => R_read(RegB()) };
let stream = Bits32(stream0) & 0x00000003; /* a word at run time, an immediate's too */
foreach (target from 0 to 3) {
    if target == 0 | MOD(DIV(StreamMask(), 2 ^ target), 2) == 1 then {
        StreamSelect(stream, target);
        if Field(39, 1) == 0b1 then {
            if target == 0 then VertexEnd() else XfbCapture(target);
            EmitVertex(target);
            if HasPointSize(true) & not_bool(PointSizeUsed()) then DefaultPointSize()
        };
        if Field(40, 1) == 0b1 then EndPrimitive(target);
        StreamEnd()
    }
};
R_write(Dst(), 0x00000000)
