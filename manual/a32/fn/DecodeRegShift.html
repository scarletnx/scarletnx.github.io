val DecodeRegShift : bits(2) -> SRType

function DecodeRegShift srtype = {
    shift_t : SRType = undefined;
    match srtype {
      0b00 => {
          shift_t = SRType_LSL
      },
      0b01 => {
          shift_t = SRType_LSR
      },
      0b10 => {
          shift_t = SRType_ASR
      },
      0b11 => {
          shift_t = SRType_ROR
      }
    };
    let shift_t = shift_t;
    return(shift_t)
}
