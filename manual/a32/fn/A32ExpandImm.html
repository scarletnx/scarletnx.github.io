val A32ExpandImm : bits(12) -> bits(32)

function A32ExpandImm imm12 = {
    imm32 : bits(32) = undefined;
    let (__tup_0 : bits(32), _) = A32ExpandImm_C(imm12, PSTATE.C) in
      {
          imm32 = __tup_0
      };
    return(imm32)
}
